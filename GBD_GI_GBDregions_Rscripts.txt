# Drop extra ID columns, enforce types, trim strings
# ============================================================

clean_regional <- function(df) {
  
  # Drop the two extra population columns (constant = "All Population")
  df <- dplyr::select(df, -population_group_id, -population_group_name)
  
  # Drop numeric ID columns — names are sufficient for analysis
  df <- dplyr::select(df, -measure_id, -sex_id, -age_id,
                      -cause_id, -metric_id, -location_id)
  
  # Rename to clean short names
  df <- dplyr::rename(df,
                      location  = location_name,
                      cause     = cause_name,
                      measure   = measure_name,
                      metric    = metric_name,
                      sex       = sex_name,
                      age_group = age_name
  )
  
  # Trim whitespace on all character columns
  df$location  <- trimws(df$location)
  df$cause     <- trimws(df$cause)
  df$measure   <- trimws(df$measure)
  df$metric    <- trimws(df$metric)
  df$sex       <- trimws(df$sex)
  df$age_group <- trimws(df$age_group)
  
  # Enforce types
  df$year  <- as.integer(df$year)
  df$val   <- as.numeric(df$val)
  df$upper <- as.numeric(df$upper)
  df$lower <- as.numeric(df$lower)
  
  return(df)
}

# Format number with thousand separator, 1 decimal
fmt_num <- function(x) formatC(round(x, 1), format = "f", digits = 1, big.mark = ",")

# Format rate to 2 decimal places
fmt_rate <- function(x) formatC(round(x, 2), format = "f", digits = 2)

# Format percent change
fmt_pct <- function(x) paste0(ifelse(x >= 0, "+", ""), round(x, 1), "%")

# Extract 1990 and 2023 values and compute % change
burden_change <- function(df, value_col = "val", ui_lo = "lower", ui_hi = "upper") {
  y1990 <- dplyr::filter(df, year == 1990)
  y2023 <- dplyr::filter(df, year == 2023)
  merged <- dplyr::inner_join(
    dplyr::select(y1990, location, cause, measure, sex,
                  val_1990 = !!sym(value_col),
                  lo_1990  = !!sym(ui_lo),
                  hi_1990  = !!sym(ui_hi)),
    dplyr::select(y2023, location, cause, measure, sex,
                  val_2023 = !!sym(value_col),
                  lo_2023  = !!sym(ui_lo),
                  hi_2023  = !!sym(ui_hi)),
    by = c("location", "cause", "measure", "sex")
  )
  merged <- dplyr::mutate(merged,
                          pct_change = ((val_2023 - val_1990) / val_1990) * 100
  )
  return(merged)
}

             rows = i + 1, cols = 1:nc, gridExpand = TRUE)
  }
  setColWidths(wb, sheet_name, cols = 1:nc, widths = col_widths)
  setRowHeights(wb, sheet_name, rows = 1, heights = 30)
}

# SCRIPT 09 — BLOCK B: EAPC Core Functions
# Log-linear regression for overall EAPC
# Segmented joinpoint for trend change detection
# ============================================================

# --- Overall EAPC via log-linear regression ---
calc_eapc <- function(years, rates) {
  tryCatch({
    if (any(is.na(rates)) || any(rates <= 0) || length(rates) < 5) {
      return(list(eapc = NA, ci_lo = NA, ci_hi = NA, pval = NA, r2 = NA))
    }
    ln_rates <- log(rates)
    fit      <- lm(ln_rates ~ years)
    beta     <- coef(fit)["years"]
    se       <- summary(fit)$coefficients["years", "Std. Error"]
    eapc     <- (exp(beta) - 1) * 100
    ci_lo    <- (exp(beta - 1.96 * se) - 1) * 100
    ci_hi    <- (exp(beta + 1.96 * se) - 1) * 100
    pval     <- summary(fit)$coefficients["years", "Pr(>|t|)"]
    r2       <- summary(fit)$r.squared
    list(eapc = eapc, ci_lo = ci_lo, ci_hi = ci_hi, pval = pval, r2 = r2)
  }, error = function(e) {
    list(eapc = NA, ci_lo = NA, ci_hi = NA, pval = NA, r2 = NA)
  })
}

# --- Joinpoint via segmented package ---
calc_joinpoint <- function(years, rates) {
  tryCatch({
    if (any(is.na(rates)) || any(rates <= 0) || length(rates) < 10) {
      return(list(n_joinpoints = 0, joinpoint_years = NA,
                  segment_eapcs = NA, segment_cis = NA))
    }
    ln_rates <- log(rates)
    fit_lm   <- lm(ln_rates ~ years)
    fit_seg  <- segmented::segmented(fit_lm,
                                     seg.Z = ~years,
                                     npsi  = 1,
                                     control = segmented::seg.control(
                                       it.max = 100, tol = 1e-5, seed = 42))
    bp       <- fit_seg$psi[, "Est."]
    bp_year  <- round(bp)
    
    # Extract slopes per segment
    slopes   <- segmented::slope(fit_seg)$years
    n_seg    <- nrow(slopes)
    seg_eapc <- round((exp(slopes[, "Est."]) - 1) * 100, 2)
    seg_lo   <- round((exp(slopes[, "Est."] - 1.96 * slopes[, "Std. Error"]) - 1) * 100, 2)
    seg_hi   <- round((exp(slopes[, "Est."] + 1.96 * slopes[, "Std. Error"]) - 1) * 100, 2)
    
    seg_str  <- paste0(seg_eapc, "% (", seg_lo, "–", seg_hi, ")",
                       collapse = "; ")
    
    list(n_joinpoints    = length(bp_year),
         joinpoint_years = paste(bp_year, collapse = ", "),
         segment_eapcs   = seg_str,
         segment_cis     = NA)
  }, error = function(e) {
    list(n_joinpoints = 0, joinpoint_years = NA,
         segment_eapcs = NA, segment_cis = NA)
  })
}

# --- Format helpers ---
fmt_eapc <- function(e, lo, hi) {
  if (is.na(e)) return(NA_character_)
  paste0(round(e, 2), "% (", round(lo, 2), "–", round(hi, 2), "%)")
}

fmt_p <- function(p) {
  if (is.na(p)) return(NA_character_)
  if (p < 0.001) return("<0.001")
  formatC(p, format = "f", digits = 3)
}

    N_joinpoints     = jp$n_joinpoints,

  )

format_eapc_table <- function(df) {
  df <- dplyr::mutate(df,
                      `EAPC (95% CI)`     = mapply(fmt_eapc, EAPC, CI_lo, CI_hi),
                      `P-value`           = sapply(P_value, fmt_p),
                      `R²`                = ifelse(is.na(R2), NA_character_,
                                                   formatC(round(R2, 3), format = "f", digits = 3)),
                      `Joinpoint Year(s)` = ifelse(is.na(Joinpoint_years), "None", Joinpoint_years),
                      `Segment EAPC(s)`   = ifelse(is.na(Segment_EAPCs),  "—",    Segment_EAPCs),
                      Trend               = dplyr::case_when(
                        is.na(EAPC)         ~ "Undetermined",
                        P_value >= 0.05     ~ "Stable",
                        EAPC > 0            ~ "Increasing",
                        TRUE                ~ "Decreasing"
                      )
  ) |>
    dplyr::select(
      Location, Condition, Measure, Sex,
      `EAPC (95% CI)`, `P-value`, `R²`, Trend,
      `Joinpoint Year(s)`, `Segment EAPC(s)`
    ) |>
    dplyr::arrange(Condition, Measure, Sex, Location) |>
    as.data.frame()
  return(df)
}

# SCRIPT 10 — BLOCK B: ARIMA Core Functions
# AICc model selection, hold-out validation 2019-2023,
# forecast 2024-2035 with 95% PI
# ============================================================

run_arima_forecast <- function(years, rates, loc, cond, meas) {
  
  tryCatch({
    
    if (length(rates) < 10 || any(is.na(rates)) || any(rates <= 0)) {
      return(NULL)
    }
    
    # --- Split: training 1990-2018, hold-out 2019-2023 ---
    train_idx  <- years <= 2018
    holdout_idx <- years >= 2019 & years <= 2023
    
    train_rates   <- rates[train_idx]
    train_years   <- years[train_idx]
    holdout_rates <- rates[holdout_idx]
    holdout_years <- years[holdout_idx]
    
    # --- Fit ARIMA on training set (AICc selection) ---
    ts_train <- ts(train_rates, start = min(train_years), frequency = 1)
    fit      <- forecast::auto.arima(ts_train, ic = "aicc",
                                     stepwise = FALSE, approximation = FALSE)
    
    # --- Validation: forecast 5 steps ahead (2019-2023) ---
    fc_holdout <- forecast::forecast(fit, h = 5, level = 95)
    val_pred   <- as.numeric(fc_holdout$mean)
    val_lo     <- as.numeric(fc_holdout$lower[, 1])
    val_hi     <- as.numeric(fc_holdout$upper[, 1])
    
    # Validation metrics
    mae  <- mean(abs(holdout_rates - val_pred))
    rmse <- sqrt(mean((holdout_rates - val_pred)^2))
    mape <- mean(abs((holdout_rates - val_pred) / holdout_rates)) * 100
    coverage <- mean(holdout_rates >= val_lo & holdout_rates <= val_hi) * 100
    
    # --- Refit on full 1990-2023 for final forecast ---
    ts_full  <- ts(rates, start = min(years), frequency = 1)
    fit_full <- forecast::auto.arima(ts_full, ic = "aicc",
                                     stepwise = FALSE, approximation = FALSE)
    fc_full  <- forecast::forecast(fit_full, h = 12, level = 95)
    
    forecast_years <- 2024:2035
    fc_mean <- as.numeric(fc_full$mean)
    fc_lo   <- as.numeric(fc_full$lower[, 1])
    fc_hi   <- as.numeric(fc_full$upper[, 1])
    
    model_order <- paste0("ARIMA(",
                          paste(arimaorder(fit_full), collapse = ","), ")")
    
    # --- Assemble validation summary row ---
    val_df <- data.frame(
      Location    = loc,
      Condition   = cond,
      Measure     = meas,
      Model       = model_order,
      AICc        = round(fit_full$aicc, 2),
      MAE         = round(mae, 4),
      RMSE        = round(rmse, 4),
      MAPE        = round(mape, 2),
      Coverage_95 = round(coverage, 1),
      stringsAsFactors = FALSE
    )
    
    # --- Assemble forecast rows ---
    fc_df <- data.frame(
      Location      = loc,
      Condition     = cond,
      Measure       = meas,
      Year          = forecast_years,
      Forecast      = round(fc_mean, 4),
      PI_lower_95   = round(fc_lo,   4),
      PI_upper_95   = round(fc_hi,   4),
      stringsAsFactors = FALSE
    )
    
    list(validation = val_df, forecast = fc_df)
    
  }, error = function(e) {
    message("ARIMA failed: ", loc, " | ", cond, " | ", meas, " — ", e$message)
    NULL
  })
}

# SCRIPT 12 — BLOCK D: SDI Correlation with EAPC
# Pearson and Spearman correlations
# ============================================================

eapc_sdi <- eapc_both |>
  dplyr::left_join(sdi_lookup, by = "Location") |>
  dplyr::mutate(
    Measure = dplyr::recode(Measure, !!!measure_labels)
  )

# Compute correlations per condition × measure
corr_results <- lapply(
  split(eapc_sdi, list(eapc_sdi$Condition, eapc_sdi$Measure)),
  function(sub) {
    if (nrow(sub) < 5 || all(is.na(sub$EAPC))) return(NULL)
    pearson  <- cor.test(sub$SDI_mean, sub$EAPC, method = "pearson")
    spearman <- cor.test(sub$SDI_mean, sub$EAPC, method = "spearman",
                         exact = FALSE)
    data.frame(
      Condition       = sub$Condition[1],
      Measure         = sub$Measure[1],
      Pearson_r       = round(pearson$estimate,  3),
      Pearson_p       = round(pearson$p.value,   4),
      Spearman_rho    = round(spearman$estimate, 3),
      Spearman_p      = round(spearman$p.value,  4),
      stringsAsFactors = FALSE
    )
  }
)

corr_table <- as.data.frame(do.call(rbind,
                                    Filter(Negate(is.null), corr_results)))
corr_table <- corr_table[order(corr_table$Condition, corr_table$Measure), ]
rownames(corr_table) <- NULL

print(corr_table)

  setRowHeights(wb, sheet_name, rows = 1, heights = 32)

# --- Step 1: EAPC on age-standardized rates (true epi change) ---
calc_eapc <- function(years, rates) {
  tryCatch({
    if (any(is.na(rates)) || any(rates <= 0) || length(rates) < 5)
      return(list(eapc = NA, ci_lo = NA, ci_hi = NA, pval = NA))
    ln_rates <- log(rates)
    fit      <- lm(ln_rates ~ years)
    beta     <- coef(fit)["years"]
    se       <- summary(fit)$coefficients["years", "Std. Error"]
    list(
      eapc  = (exp(beta) - 1) * 100,
      ci_lo = (exp(beta - 1.96 * se) - 1) * 100,
      ci_hi = (exp(beta + 1.96 * se) - 1) * 100,
      pval  = summary(fit)$coefficients["years", "Pr(>|t|)"]
    )
  }, error = function(e)
    list(eapc = NA, ci_lo = NA, ci_hi = NA, pval = NA))
}

# --- Step 2: EAPC on all-ages counts (population growth + epi) ---
# Use both sexes, all-ages, Number metric
combined_counts <- regional_counts_both |>
  dplyr::filter(sex == "Both")

groups_cnt <- combined_counts |>
  dplyr::group_by(location, cause, measure) |>
  dplyr::group_split()

eapc_cnt_list <- lapply(groups_cnt, function(s) {
  s    <- s[order(s$year), ]
  res  <- calc_eapc(s$year, s$val)
  data.frame(
    Location  = as.character(s$location[1]),
    Condition = as.character(s$cause[1]),
    Measure   = as.character(s$measure[1]),
    EAPC_cnt  = res$eapc,
    CI_lo_cnt = res$ci_lo,
    CI_hi_cnt = res$ci_hi,
    Pval_cnt  = res$pval,
    stringsAsFactors = FALSE
  )
})
eapc_counts_df <- as.data.frame(do.call(rbind, eapc_cnt_list))

fmt_eapc_ci <- function(e, lo, hi) {
  if (is.na(e)) return(NA_character_)
  paste0(round(e, 2), "% (", round(lo, 2), "–", round(hi, 2), "%)")
}

fmt_p <- function(p) {
  if (is.na(p)) return(NA_character_)
  if (p < 0.001) return("<0.001")
  formatC(p, format = "f", digits = 3)
}

# Interpretation logic (mirrors global/SDI approach):
# ASR declining + Count declining  → True burden reduction
# ASR declining + Count stable/inc → Population growth masking real improvement
# ASR stable    + Count increasing → Pure population growth effect
# ASR increasing + Count increasing → True burden increase
# ASR increasing + Count stable    → Ageing effect without incidence rise
# ASR stable + Count stable        → No meaningful change

combined_eapc <- combined_eapc |>
  dplyr::mutate(
    ASR_direction = dplyr::case_when(
      is.na(EAPC_asr)  ~ "Undetermined",
      Pval_asr >= 0.05 ~ "Stable",
      EAPC_asr > 0     ~ "Increasing",
      TRUE             ~ "Decreasing"
    ),
    Cnt_direction = dplyr::case_when(
      is.na(EAPC_cnt)  ~ "Undetermined",
      Pval_cnt >= 0.05 ~ "Stable",
      EAPC_cnt > 0     ~ "Increasing",
      TRUE             ~ "Decreasing"
    ),
    Interpretation = dplyr::case_when(
      ASR_direction == "Decreasing" & Cnt_direction == "Decreasing" ~
        "True burden reduction",
      ASR_direction == "Decreasing" & Cnt_direction == "Stable" ~
        "Epidemiological improvement; stable absolute burden",
      ASR_direction == "Decreasing" & Cnt_direction == "Increasing" ~
        "Epidemiological improvement offset by population growth",
      ASR_direction == "Stable"     & Cnt_direction == "Increasing" ~
        "Population growth effect; no true rate change",
      ASR_direction == "Stable"     & Cnt_direction == "Stable" ~
        "No meaningful change",
      ASR_direction == "Increasing" & Cnt_direction == "Increasing" ~
        "True burden increase",
      ASR_direction == "Increasing" & Cnt_direction == "Stable" ~
        "Rate increasing; population decline offsetting counts",
      ASR_direction == "Increasing" & Cnt_direction == "Decreasing" ~
        "Rate increasing; population decline masking count reduction",
      TRUE ~ "Undetermined"
    )

# ── Forecast data (2024-2035) ────────────────────────────────
fc_data <- arima_results %>%
  dplyr::mutate(
    period   = "Forecast",
    value    = forecast,
    value_lo = lower_95,
    value_hi = upper_95,
    cause_short = factor(cause_short,
                         levels = c("Appendicitis","Hernia","Gallbladder")),
    measure = factor(measure,
                     levels = c("Incidence","Deaths",
                                "DALYs (Disability-Adjusted Life Years)"))
  ) %>%
  dplyr::select(location, cause_short, measure,
                year, value, value_lo, value_hi, period)

# ── Connector point (last observed year = 2023) ──────────────
connector <- obs_data %>%
  dplyr::filter(year == 2023) %>%
  dplyr::mutate(period = "Forecast")

# ── Combine ──────────────────────────────────────────────────
fig_data <- bind_rows(obs_data, connector, fc_data) %>%
  dplyr::mutate(
    measure_label = dplyr::recode(as.character(measure),
                                  "Incidence"                              = "Incidence",
                                  "Deaths"                                 = "Mortality",
                                  "DALYs (Disability-Adjusted Life Years)" = "DALYs"
    ),
    measure_label = factor(measure_label,
                           levels = c("Incidence","Mortality","DALYs")),
    panel = paste0(as.character(cause_short), "\n", measure_label),
    panel = factor(panel, levels = c(
      "Appendicitis\nIncidence",  "Appendicitis\nMortality",  "Appendicitis\nDALYs",
      "Hernia\nIncidence",        "Hernia\nMortality",        "Hernia\nDALYs",
      "Gallbladder\nIncidence",   "Gallbladder\nMortality",   "Gallbladder\nDALYs"
    )),
    location = factor(location,
                      levels = c("Global","Low SDI","Low-middle SDI",
                                 "Middle SDI","High-middle SDI","High SDI")),
    # Dashed line for forecast period
    line_type = ifelse(period == "Forecast", "Forecast", "Observed")
  )

# ── PI ribbons for forecast (Global only to avoid clutter) ───
fc_ribbon <- fig_data %>%
  dplyr::filter(period == "Forecast", location == "Global")

obs_ribbon <- fig_data %>%
  dplyr::filter(period == "Observed", location == "Global")

# ── Plot ─────────────────────────────────────────────────────
fig_arima_sdi <- ggplot(fig_data,
                        aes(x         = year,
                            y         = value,
                            color     = location,
                            linewidth = location,
                            linetype  = line_type)) +
  
  # Global 95% UI ribbon (observed) — x explicit
  geom_ribbon(
    data = obs_ribbon,
    aes(x = year, ymin = value_lo, ymax = value_hi, fill = location),
    alpha = 0.10, color = NA, inherit.aes = FALSE
  ) +
  
  # Global 95% PI ribbon (forecast) — x explicit
  geom_ribbon(
    data = fc_ribbon,
    aes(x = year, ymin = value_lo, ymax = value_hi, fill = location),
    alpha = 0.18, color = NA, inherit.aes = FALSE
  ) +
  
  geom_line() +
  
  geom_vline(xintercept = 2023, linetype = "dotted",
             color = "grey50", linewidth = 0.45) +
  
  facet_wrap(~ panel, scales = "free", ncol = 3) +
  
  scale_color_manual(values = sdi_colors, name = "Location") +
  scale_fill_manual(values  = sdi_colors, guide = "none") +
  scale_linewidth_manual(values = sdi_sizes, guide = "none") +
  scale_linetype_manual(
    values = c("Observed" = 1, "Forecast" = 2),
    name   = "Period"
  ) +
  scale_x_continuous(
    breaks = c(1990, 2000, 2010, 2023, 2030, 2035),
    labels = c("1990","2000","2010","2023","2030","2035")
  ) +
  scale_y_continuous(labels = scales::comma) +
  
  labs(
    title    = "Figure. Age-standardized rates by SDI level: observed 1990–2023 and ARIMA forecast 2024–2035",
    subtitle = "Both sexes; dashed lines = forecast period; shaded area = Global 95% UI/PI; dotted line = projection start",
    x        = "Year",
    y        = "Age-standardized rate per 100,000"
  ) +
  theme_gbd() +
  theme(axis.text.x = element_text(angle = 30, hjust = 1)) +
  guides(
    color    = guide_legend(nrow = 1,
                            override.aes = list(linewidth = 1,
                                                linetype  = 1)),
    linetype = guide_legend(nrow = 1)
  )

ggsave(
  file.path(FIGURES_DIR, "Fig_SDI_ARIMA_Forecast.png"),
  fig_arima_sdi,
  width  = 13,
  height = 10,
  dpi    = 300,
  bg     = "white"
)
