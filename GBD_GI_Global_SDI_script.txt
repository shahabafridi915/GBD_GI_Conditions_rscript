# ── Helper: format numbers nicely ───────────────────────────
fmt_n  <- function(x) formatC(round(x, 0), format = "f", digits = 0, big.mark = ",")
fmt_r  <- function(x) formatC(round(x, 2), format = "f", digits = 2)
fmt_ui <- function(val, lo, hi) paste0(fmt_r(val), " (", fmt_r(lo), "–", fmt_r(hi), ")")
fmt_pct <- function(x) paste0(ifelse(x >= 0, "+", ""), round(x, 1), "%")

# ============================================================
# TABLE 1 — Global Burden Summary
# Rows: cause × measure
# Cols: 1990 value+UI | 2023 value+UI | absolute change | % change
# Both sexes combined, age-standardized RATES + all-ages COUNTS
# ============================================================

make_table1 <- function() {
  
  # --- Age-standardized rates (Global, Both sexes) ---
  rates <- df_asr %>%
    filter(location == "Global", sex == "Both") %>%
    filter(year %in% c(1990, 2023)) %>%
    select(cause_short, measure, year, value, lower, upper) %>%
    pivot_wider(
      names_from  = year,
      values_from = c(value, lower, upper)
    ) %>%
    mutate(
      metric_type    = "Age-standardized rate",
      val_1990       = fmt_ui(value_1990, lower_1990, upper_1990),
      val_2023       = fmt_ui(value_2023, lower_2023, upper_2023),
      abs_change     = fmt_r(value_2023 - value_1990),
      pct_change     = fmt_pct((value_2023 - value_1990) / value_1990 * 100)
    ) %>%
    select(cause_short, measure, metric_type,
           val_1990, val_2023, abs_change, pct_change)
  
  # --- All-ages counts (Global, Both sexes) ---
  cnts <- df_counts %>%
    filter(location == "Global", sex == "Both") %>%
    filter(year %in% c(1990, 2023)) %>%
    select(cause_short, measure, year, value, lower, upper) %>%
    pivot_wider(
      names_from  = year,
      values_from = c(value, lower, upper)
    ) %>%
    mutate(
      metric_type = "Absolute count",
      val_1990    = paste0(fmt_n(value_1990),
                           " (", fmt_n(lower_1990), "–", fmt_n(upper_1990), ")"),
      val_2023    = paste0(fmt_n(value_2023),
                           " (", fmt_n(lower_2023), "–", fmt_n(upper_2023), ")"),
      abs_change  = fmt_n(value_2023 - value_1990),
      pct_change  = fmt_pct((value_2023 - value_1990) / value_1990 * 100)
    ) %>%
    select(cause_short, measure, metric_type,
           val_1990, val_2023, abs_change, pct_change)
  
  bind_rows(rates, cnts) %>%
    arrange(cause_short, measure, metric_type) %>%
    rename(
      Condition        = cause_short,
      Measure          = measure,
      `Metric type`    = metric_type,
      `1990 (95% UI)`  = val_1990,
      `2023 (95% UI)`  = val_2023,
      `Absolute change` = abs_change,
      `% change`       = pct_change
    )
}

table1 <- make_table1()
cat("=== TABLE 1: Global Burden Summary ===\n")
print(table1, n = Inf)

# ============================================================
# TABLE 4 — DALY Decomposition
# Decompose absolute DALY count change (1990→2023) into:
#   Population growth effect
#   Population aging effect  (approximated)
#   Epidemiological (rate) effect
# Method: standard GBD three-component decomposition
# Using: all-ages count + age-standardized rate from df_asr/df_counts
# ============================================================

make_table4 <- function() {
  
  # Pull DALY counts and rates for 1990 and 2023
  daly_counts <- df_counts %>%
    filter(
      measure == "DALYs (Disability-Adjusted Life Years)",
      sex == "Both"
    ) %>%
    filter(year %in% c(1990, 2023)) %>%
    select(location, cause_short, year, value) %>%
    pivot_wider(names_from = year, values_from = value,
                names_prefix = "count_")
  
  daly_rates <- df_asr %>%
    filter(
      measure == "DALYs (Disability-Adjusted Life Years)",
      sex == "Both"
    ) %>%
    filter(year %in% c(1990, 2023)) %>%
    select(location, cause_short, year, value) %>%
    pivot_wider(names_from = year, values_from = value,
                names_prefix = "rate_")
  
  # GBD decomposition:
  # Total change = count_2023 - count_1990
  # Rate effect  = (rate_2023 - rate_1990) / rate_1990 * count_1990
  # Pop effect   = total_change - rate_effect  (residual = growth + aging)
  # Note: full aging/growth split requires population pyramids not in this
  # dataset; we report combined population effect and rate effect per
  # GBD standard for SDI-level analyses.
  
  daly_counts %>%
    left_join(daly_rates, by = c("location", "cause_short")) %>%
    mutate(
      total_change   = count_2023 - count_1990,
      rate_effect    = (rate_2023 - rate_1990) / rate_1990 * count_1990,
      pop_effect     = total_change - rate_effect,
      pct_rate       = round(rate_effect / abs(total_change) * 100, 1),
      pct_pop        = round(pop_effect  / abs(total_change) * 100, 1),
      `Total DALY change`       = fmt_n(total_change),
      `Population effect (n)`   = fmt_n(pop_effect),
      `Population effect (%)`   = paste0(pct_pop, "%"),
      `Epidemiological effect (n)` = fmt_n(rate_effect),
      `Epidemiological effect (%)` = paste0(pct_rate, "%")
    ) %>%
    arrange(location, cause_short) %>%
    select(location, cause_short,
           `Total DALY change`,
           `Population effect (n)`, `Population effect (%)`,
           `Epidemiological effect (n)`, `Epidemiological effect (%)`) %>%
    rename(
      Location  = location,
      Condition = cause_short
    )
}

table4 <- make_table4()

# ============================================================
# CORE EAPC FUNCTION
# ln(rate) = α + β × year + ε
# EAPC = (exp(β) - 1) × 100
# 95% CI from regression SE
# Significant if 95% CI excludes zero
# ============================================================

calc_eapc <- function(years, rates) {
  tryCatch({
    df_fit  <- data.frame(y = log(rates), x = years)
    fit     <- lm(y ~ x, data = df_fit)
    beta    <- coef(fit)["x"]
    se_beta <- summary(fit)$coefficients["x", "Std. Error"]
    p_val   <- summary(fit)$coefficients["x", "Pr(>|t|)"]
    
    eapc    <- (exp(beta) - 1) * 100
    ci_lo   <- (exp(beta - 1.96 * se_beta) - 1) * 100
    ci_hi   <- (exp(beta + 1.96 * se_beta) - 1) * 100
    r2      <- summary(fit)$r.squared
    
    list(
      eapc   = round(eapc,  3),
      ci_lo  = round(ci_lo, 3),
      ci_hi  = round(ci_hi, 3),
      p_val  = round(p_val, 4),
      r2     = round(r2,    4),
      sig    = p_val < 0.05,
      trend  = case_when(
        p_val >= 0.05              ~ "Stable",
        eapc  >  0                 ~ "Increasing",
        TRUE                       ~ "Decreasing"
      )
    )
  }, error = function(e) {
    list(eapc = NA, ci_lo = NA, ci_hi = NA,
         p_val = NA, r2 = NA, sig = NA, trend = "Error")
  })
}

cat("Running EAPC calculations...\n")

eapc_results <- df_asr %>%
  group_by(location, cause_short, measure, sex) %>%
  arrange(year, .by_group = TRUE) %>%
  summarise(
    {
      res <- calc_eapc(year, value)
      tibble(
        eapc  = res$eapc,
        ci_lo = res$ci_lo,
        ci_hi = res$ci_hi,
        p_val = res$p_val,
        r2    = res$r2,
        sig   = res$sig,
        trend = res$trend
      )
    },
    .groups = "drop"
  )

joinpoint_results <- df_asr %>%
  filter(location == "Global", sex == "Both") %>%
  group_by(cause_short, measure) %>%
  arrange(year, .by_group = TRUE) %>%
  group_modify(~ {
    yr  <- .x$year
    rt  <- log(.x$value)
    df_jp <- data.frame(y = rt, x = yr)
    
    tryCatch({
      fit_lin <- lm(y ~ x, data = df_jp)
      
      # Try up to 2 joinpoints; select by BIC
      best_fit  <- fit_lin
      best_bic  <- BIC(fit_lin)
      best_npsi <- 0
      
      for (npsi in 1:2) {
        jp_fit <- tryCatch(
          segmented(fit_lin, seg.Z = ~x,
                    npsi = npsi,
                    control = seg.control(it.max = 100, quiet = TRUE)),
          error = function(e) NULL
        )
        if (!is.null(jp_fit)) {
          b <- BIC(jp_fit)
          if (b < best_bic) {
            best_bic  <- b
            best_fit  <- jp_fit
            best_npsi <- npsi
          }
        }
      }
      
      if (best_npsi == 0) {
        tibble(
          n_joinpoints  = 0,
          breakpoints   = "None",
          interpretation = "Single trend throughout 1990–2023"
        )
      } else {
        bps <- round(best_fit$psi[, "Est."], 1)
        tibble(
          n_joinpoints   = best_npsi,
          breakpoints    = paste(bps, collapse = ", "),
          interpretation = paste0(
            best_npsi, " trend change point(s) detected at: ",
            paste(bps, collapse = " and ")
          )
        )
      }
    }, error = function(e) {
      tibble(n_joinpoints = NA, breakpoints = "Error",
             interpretation = as.character(e$message))
    })
  }) %>%
  ungroup()

# ============================================================
# STEP 2 — ARIMA forecasting function
# Hold-out validation: train on 1990-2018, test on 2019-2023
# Forecast: 2024-2035 (12 years)
# Applied to: all locations × causes × measures, Both sexes
# ============================================================

run_arima <- function(yr, vals, loc, cause, meas) {
  
  tryCatch({
    
    # Full series as ts object
    ts_full  <- ts(vals, start = min(yr), frequency = 1)
    
    # --- Hold-out validation (train 1990-2018, test 2019-2023) ---
    train_idx <- yr <= 2018
    test_idx  <- yr >= 2019
    
    ts_train <- ts(vals[train_idx],
                   start = min(yr), frequency = 1)
    
    fit_val  <- auto.arima(ts_train,
                           stepwise   = FALSE,
                           approximation = FALSE,
                           ic         = "aicc")
    
    n_test   <- sum(test_idx)
    fc_val   <- forecast(fit_val, h = n_test)
    actuals  <- vals[test_idx]
    
    mae  <- mean(abs(actuals - as.numeric(fc_val$mean)))
    rmse <- sqrt(mean((actuals - as.numeric(fc_val$mean))^2))
    mape <- mean(abs((actuals - as.numeric(fc_val$mean)) /
                       actuals)) * 100
    
    # --- Final model on full series ---
    fit_full <- auto.arima(ts_full,
                           stepwise      = FALSE,
                           approximation = FALSE,
                           ic            = "aicc")
    
    model_order <- paste0("ARIMA(",
                          paste(arimaorder(fit_full), collapse = ","), ")")
    
    # --- Forecast 2024-2035 ---
    fc <- forecast(fit_full, h = 12, level = 95)
    
    fc_df <- data.frame(
      location    = loc,
      cause_short = cause,
      measure     = meas,
      year        = 2024:2035,
      forecast    = as.numeric(fc$mean),
      lower_95    = as.numeric(fc$lower[, 1]),
      upper_95    = as.numeric(fc$upper[, 1]),
      model       = model_order,
      mae         = round(mae,  3),
      rmse        = round(rmse, 3),
      mape        = round(mape, 2),
      stringsAsFactors = FALSE
    )
    
    fc_df
    
  }, error = function(e) {
    data.frame(
      location = loc, cause_short = cause, measure = meas,
      year = 2024:2035, forecast = NA, lower_95 = NA,
      upper_95 = NA, model = "Error", mae = NA,
      rmse = NA, mape = NA,
      stringsAsFactors = FALSE
    )
  })
}

fmt_n   <- function(x) formatC(round(x,0), format="f", digits=0, big.mark=",")
fmt_pct <- function(x) paste0(ifelse(x>=0,"+",""), round(x,1), "%")

tableS1 <- df_age %>%
  dplyr::filter(location == "Global", sex == "Both") %>%
  dplyr::filter(year %in% c(1990, 2023)) %>%

                             " (", fmt_n(lower_1990),

    `2023 (95% UI)` = paste0(fmt_n(value_2023),
                             " (", fmt_n(lower_2023),
                             "–", fmt_n(upper_2023), ")"),
    `% change`      = fmt_pct((value_2023 - value_1990) /
                                value_1990 * 100)
  ) %>%
  dplyr::arrange(cause_short, measure, age) %>%
  dplyr::select(
    Condition = cause_short,
    Measure   = measure,
    `Age group` = age,
    `1990 (95% UI)`,
    `2023 (95% UI)`,
    `% change`
  )

# ============================================================
# TABLE S6 — Combined EAPC for total GI burden
# Log-linear regression on summed all-condition counts
# ============================================================

calc_eapc <- function(years, rates) {
  tryCatch({
    fit   <- lm(log(rates) ~ years)
    beta  <- coef(fit)["years"]
    se    <- summary(fit)$coefficients["years","Std. Error"]
    pval  <- summary(fit)$coefficients["years","Pr(>|t|)"]
    list(
      eapc  = round((exp(beta)-1)*100, 3),
      ci_lo = round((exp(beta-1.96*se)-1)*100, 3),
      ci_hi = round((exp(beta+1.96*se)-1)*100, 3),
      pval  = round(pval, 4),
      r2    = round(summary(fit)$r.squared, 4)
    )
  }, error = function(e)
    list(eapc=NA, ci_lo=NA, ci_hi=NA, pval=NA, r2=NA)
  )
}

# Use age-standardized combined rate (mean across conditions
# is not valid — use sum of counts divided by sum approach)
# For EAPC we use the all-ages combined count trend
combined_eapc_list <- combined_counts %>%
  dplyr::group_by(location, measure) %>%
  dplyr::group_split() %>%
  lapply(function(s) {
    loc  <- as.character(s$location[1])
    meas <- as.character(s$measure[1])
    res  <- calc_eapc(s$year, s$combined_value)
    data.frame(
      Location  = loc,
      Measure   = meas,
      `EAPC`    = res$eapc,
      `CI_lo`   = res$ci_lo,
      `CI_hi`   = res$ci_hi,
      `P value` = ifelse(is.na(res$pval), NA,
                         ifelse(res$pval < 0.001, "<0.001",
                                formatC(res$pval, format="f", digits=3))),
      `R2`      = res$r2,
      `EAPC (95% CI)` = ifelse(is.na(res$eapc), "Error",
                               paste0(formatC(res$eapc, format="f", digits=2),
                                      " (", formatC(res$ci_lo, format="f", digits=2),
                                      " to ", formatC(res$ci_hi, format="f", digits=2), ")")),
      stringsAsFactors = FALSE,
      check.names = FALSE
    )
  })

tableS6 <- bind_rows(combined_eapc_list) %>%
  dplyr::mutate(
    Location = factor(Location,
                      levels = c("Global","Low SDI","Low-middle SDI",
                                 "Middle SDI","High-middle SDI","High SDI")),
    Significant = ifelse(is.na(`P value`), "Unknown",
                         ifelse(`P value` == "<0.001" |
                                  as.numeric(`P value`) < 0.05,
                                "Yes", "No"))
  ) %>%
  dplyr::arrange(Location, Measure) %>%
  dplyr::select(Location, Measure, `EAPC (95% CI)`,
                `P value`, R2, Significant)
